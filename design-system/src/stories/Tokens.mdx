# Tokens

Every value below is rendered by referencing the live CSS custom property (`var(--token)`)
declared in `src/tokens.css` — nothing here is a hardcoded copy of a color, size, or family
that could drift out of sync with the source. Flip the theme in the toolbar above (or add
`?globals=theme:light` to the URL) and the swatches update immediately, because they resolve
the same `:root[data-theme]` cascade every other component does.

export function useLiveVar(name) {
  const [value, setValue] = useState("")

  useEffect(() => {
    const read = () => {
      const resolved = getComputedStyle(document.documentElement).getPropertyValue(name).trim()
      setValue(resolved)
    }
    read()

    const observer = new MutationObserver(read)
    observer.observe(document.documentElement, { attributes: true, attributeFilter: ["data-theme"] })
    return () => observer.disconnect()
  }, [name])

  return value
}

export function ColorSwatch({ name, description }) {
  const resolved = useLiveVar(name)
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 12, padding: "8px 4px" }}>
      <div
        style={{
          width: 40,
          height: 40,
          flexShrink: 0,
          borderRadius: "var(--rounded-md)",
          background: `var(${name})`,
          border: "1px solid var(--border-strong)",
        }}
      />
      <div style={{ minWidth: 0 }}>
        <div style={{ fontFamily: "var(--ff-num)", fontWeight: 600, fontSize: "0.8125rem", color: "var(--ink)" }}>
          {name}
        </div>
        <div style={{ fontFamily: "var(--ff-num)", fontSize: "0.75rem", color: "var(--muted)" }}>
          {resolved || "…"}
        </div>
        {description && (
          <div style={{ fontSize: "0.8125rem", color: "var(--faint)", marginTop: 2 }}>{description}</div>
        )}
      </div>
    </div>
  )
}

export function ColorGrid({ tokens }) {
  return (
    <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(240px, 1fr))", gap: 4 }}>
      {tokens.map(([name, description]) => (
        <ColorSwatch key={name} name={name} description={description} />
      ))}
    </div>
  )
}

export function ScaleRow({ name, dimension = "width", swatch = "background" }) {
  const resolved = useLiveVar(name)
  const boxStyle =
    swatch === "background"
      ? {
          [dimension]: `var(${name})`,
          height: dimension === "width" ? 24 : `var(${name})`,
          background: "var(--accent)",
          borderRadius: "var(--rounded-sm)",
        }
      : {
          width: 56,
          height: 56,
          background: "var(--surface-2)",
          border: "1px solid var(--border-strong)",
          borderRadius: `var(${name})`,
        }

  return (
    <div style={{ display: "flex", alignItems: "center", gap: 16, padding: "6px 0" }}>
      <code style={{ width: 130, flexShrink: 0, fontSize: "0.8125rem", color: "var(--ink)" }}>{name}</code>
      <div style={{ ...boxStyle, flexShrink: 0 }} />
      <span style={{ fontFamily: "var(--ff-num)", fontSize: "0.8125rem", color: "var(--muted)" }}>
        {resolved || "…"}
      </span>
    </div>
  )
}

export function ZIndexRow({ name, description }) {
  const resolved = useLiveVar(name)
  return (
    <tr>
      <td style={{ padding: "6px 12px 6px 0", fontFamily: "var(--ff-num)", fontSize: "0.8125rem", color: "var(--ink)" }}>
        {name}
      </td>
      <td style={{ padding: "6px 12px 6px 0", fontFamily: "var(--ff-num)", fontSize: "0.8125rem", color: "var(--muted)" }}>
        {resolved || "…"}
      </td>
      <td style={{ padding: "6px 0", fontSize: "0.8125rem", color: "var(--faint)" }}>{description}</td>
    </tr>
  )
}

export function TypeSpecimen({ family, label, sample }) {
  const resolved = useLiveVar(family)
  return (
    <div style={{ padding: "10px 0", borderBottom: "1px solid var(--border)" }}>
      <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 4 }}>
        <code style={{ fontSize: "0.8125rem", color: "var(--ink)" }}>{family}</code>
        <span style={{ fontFamily: "var(--ff-num)", fontSize: "0.75rem", color: "var(--faint)" }}>{label}</span>
      </div>
      <div style={{ fontFamily: `var(${family})`, fontSize: "1.5rem", color: "var(--ink)" }}>{sample}</div>
      <div style={{ fontFamily: "var(--ff-num)", fontSize: "0.75rem", color: "var(--muted)", marginTop: 2 }}>
        {resolved || "…"}
      </div>
    </div>
  )
}

## Color roles

The semantic role layer components actually consume — resolved per `data-theme`. Dark-theme
values are byte-for-byte restatements of the original marketing palette (see DESIGN.md's
"Dark-Truth Rule"); light theme is new territory built from the same brand ramp.

### State vocabulary

Every interactive surface speaks the same state words — danger, warning, success, info —
each with a base, a low-alpha `-bg` tint for chip/banner fills, and an `-ink` for text/icon
on that tint or directly on the page surface.

## Typography

Only the family tokens are exposed as CSS custom properties today; the fixed-rem type
*scale* itself (Display/Headline/Title/Body/Label/Caption steps) is a documented convention
in `DESIGN.md`, not yet a set of CSS variables — so it's intentionally not swatched here as
if it were one.
